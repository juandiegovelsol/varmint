+        DrawRecorder batch = new DrawRecorder();

+            renderer.draw(batch, zLevel, view);
+        }
+
+        String what = "heading " + heading + "°: ";
+        List<DrawRecorder.Quad> bodies = new ArrayList<>();
+        for (DrawRecorder.Quad quad : batch.quads()) {
+            if (!quad.isShadowLike()) {

+                            + " every other shadow; it is offset by "
+                            + offset,

+                    what + "the shadow of a car on the ground should stay close to it: " + offset,
+                    offset.len() < 1);
+        }
+    }

+/*
+ * How the race view follows the player's car: it keeps the car on screen and leaves more room in
+ * front of the car than behind it, whichever way the car is heading.
+ *
+ * The camera updater used by the race screen is driven with a car standing still in the middle of
+ * a large track, facing various directions, and the resulting camera is inspected.
+ */
+package com.agateau.pixelwheels.racescreen;
+
+import static org.junit.Assert.assertTrue;
+import static org.mockito.Mockito.mock;
+import static org.mockito.Mockito.when;
+
+import com.agateau.pixelwheels.GameWorld;
+import com.agateau.pixelwheels.map.Track;
+import com.agateau.pixelwheels.racer.Racer;
+import com.agateau.pixelwheels.racer.Vehicle;
+import com.badlogic.gdx.graphics.OrthographicCamera;
+import com.badlogic.gdx.math.Vector2;
+import org.junit.Test;
+import org.junit.runner.RunWith;
+import org.junit.runners.JUnit4;
+
+@RunWith(JUnit4.class)
+public class RacerCameraTest {
+    private static final float[] HEADINGS = {0, 45, 90, 135, 180, 225, 270, 315};
+    private static final Vector2 CAR = new Vector2(500, 500);
+
+    /** Returns where the view is centered, relative to the car. */
+    private static Vector2 viewOffset(float heading) {
+        Track track = mock(Track.class);
+        when(track.getMapWidth()).thenReturn(1000f);
+        when(track.getMapHeight()).thenReturn(1000f);
+        GameWorld world = mock(GameWorld.class);
+        when(world.getTrack()).thenReturn(track);
+
+        Vehicle vehicle = mock(Vehicle.class);
+        when(vehicle.getPosition()).thenReturn(new Vector2(CAR));
+        when(vehicle.getX()).thenReturn(CAR.x);
+        when(vehicle.getY()).thenReturn(CAR.y);
+        when(vehicle.getAngle()).thenReturn(heading);
+        when(vehicle.getSpeed()).thenReturn(0f);
+        Racer racer = mock(Racer.class);
+        when(racer.getVehicle()).thenReturn(vehicle);
+        when(racer.getCameraAngle()).thenReturn(heading);
+        when(racer.getPosition()).thenReturn(new Vector2(CAR));
+        when(racer.getX()).thenReturn(CAR.x);
+        when(racer.getY()).thenReturn(CAR.y);
+
+        OrthographicCamera camera = new OrthographicCamera();
+        RacerCameraUpdater updater = new RacerCameraUpdater(world, racer);
+        updater.init(camera, 1920, 1080);
+        updater.update(CameraUpdater.IMMEDIATE);
+
+        float halfW = camera.viewportWidth * camera.zoom / 2;
+        float halfH = camera.viewportHeight * camera.zoom / 2;
+        Vector2 offset = new Vector2(camera.position.x, camera.position.y).sub(CAR);
+        assertTrue(
+                "heading " + heading + "°: the car is out of the view (view offset " + offset + ")",
+                Math.abs(offset.x) < halfW && Math.abs(offset.y) < halfH);
+        return offset;
+    }
+
+    private static float angleBetween(Vector2 direction, float angleDeg) {
+        float delta = (direction.angleDeg() - angleDeg) % 360;
+        if (delta < -180) {
+            delta += 360;
+        } else if (delta > 180) {
+            delta -= 360;
+        }
+        return Math.abs(delta);
+    }
+
+    @Test
+    public void viewLooksAheadOfTheCarWhicheverWayItHeads() {
+        for (float heading : HEADINGS) {
+            Vector2 offset = viewOffset(heading);
+            assertTrue(
+                    "heading " + heading + "°: the view should show more of the road ahead of the"
+                            + " car than behind it (view offset " + offset + ")",
+                    offset.len() > 1);
+            assertTrue(
+                    "heading " + heading + "°: the view is shifted towards " + offset.angleDeg()
+                            + "° instead of towards where the car is heading",
+                    angleBetween(offset, heading) < 5);
+        }
+    }
+
+    @Test
+    public void viewKeepsTheCarOnScreen() {
+        for (float heading : HEADINGS) {
+            viewOffset(heading);
+        }
+    }
