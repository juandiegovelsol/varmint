+ * Reads how a rendered grid header actually looks, for the header-row specs.
+ *
+ * jsdom loads no stylesheet, so a class name alone says nothing about the look. This helper builds the
+ * page's real stylesheet the way the app ships it — the ui-kit and file-manager styles, then the app's own
+ * `global.scss` compiled by Sass and Tailwind — and runs a small CSS cascade (layers, `!important`,
+ * specificity, source order, inline styles, inheritance, `var()`) over the rendered DOM. A header passes
+ * or fails on the look it ends up with, whether that comes from a utility class, an arbitrary value, an
+ * app stylesheet rule, an `@utility`, a kit class or an inline style.

+ * header specs depend on (its right-aligned column type reverses the label row) is added by hand.

+const compareSpecificity = (a: Specificity, b: Specificity) => a[0] - b[0] || a[1] - b[1] || a[2] - b[2];

+const splitTopLevel = (list: string): string[] => {
+  const parts: string[] = [];

+  let current = '';
+  for (let i = 0; i < list.length; i++) {
+    const char = list[i];
+    if (char === '\\') {
+      current += char + (list[i + 1] ?? '');
+      i++;

+    if (char === ')' || char === ']') depth--;
+    if (char === ',' && depth === 0) {
+      parts.push(current);
+      current = '';
+    } else {

+    }

+  let result: Specificity = [0, 0, 0];
+  let i = 0;

+  };

+    let depth = 0;
+    const start = i;
+    for (; i < selector.length; i++) {
+      if (selector[i] === '\\') {
+        i++;
+        continue;
+      }
+      if (selector[i] === '(') depth++;
+      if (selector[i] === ')' && --depth === 0) {
+        i++;
+        break;
+      }
+    }
+    return selector.slice(start + 1, i - 1);
+  };
+  while (i < selector.length) {
+    const char = selector[i];
+    if (char === '#') {
+      i++;
+      readIdent();
+      result = addSpecificity(result, [1, 0, 0]);
+    } else if (char === '.') {
+      i++;
+      readIdent();
+      result = addSpecificity(result, [0, 1, 0]);
+    } else if (char === '[') {
+      while (i < selector.length && selector[i] !== ']') i += selector[i] === '\\' ? 2 : 1;
+      i++;
+      result = addSpecificity(result, [0, 1, 0]);
+    } else if (char === ':') {
+      const element = selector[i + 1] === ':';
+      i += element ? 2 : 1;
+      const start = i;
+      readIdent();
+      const name = selector.slice(start, i).toLowerCase();

+      if (element) {
+        result = addSpecificity(result, [0, 0, 1]);

+          .reduce<Specificity>((max, next) => (compareSpecificity(next, max) > 0 ? next : max), [0, 0, 0]);
+        result = addSpecificity(result, most);

+      readIdent();
+      result = addSpecificity(result, [0, 0, 1]);
+    } else {
+      i++;
+    }
+  }
+  return result;

+  /** Cascade rank, compared left to right; higher wins. */
+  rank: number[];

+const matchingSpecificity = (element: Element, selectors: string[]): Specificity | null => {
+  let best: Specificity | null = null;
+  for (const selector of selectors) {
+    if (selector.includes('::')) continue;
+    let matches = false;
+    try {
+      matches = element.matches(selector);
+    } catch {
+      matches = false;
+    }
+    if (!matches) continue;
+    const own = specificity(selector);
+    if (!best || compareSpecificity(own, best) > 0) best = own;
+  }
+  return best;
+};

+const matchedRules = new WeakMap<Element, { rule: StyleRule; specificity: Specificity }[]>();

+    matched = [];
+    for (const rule of styles.rules) {
+      const found = matchingSpecificity(element, rule.selectors);
+      if (found) matched.push({ rule, specificity: found });
+    }
+    matchedRules.set(element, matched);
+  }
+  return matched;

+/** The value the cascade gives `property` on this element itself, before inheritance. */
+const cascadedValue = (styles: PageStyles, element: Element, property: string): string | undefined => {
+  const candidates: Candidate[] = [];

+    const declaration = rule.declarations.get(property);

+    // the layer order and beats every normal declaration.

+        : styles.layerCount - rule.layer

+    candidates.push({ value: inlineValue.trim(), important, rank: [important ? 3 : 1] });
+  }
+  const winner = candidates.sort((a, b) => {
+    for (let i = 0; i < Math.max(a.rank.length, b.rank.length); i++) {
+      const delta = (a.rank[i] ?? 0) - (b.rank[i] ?? 0);
+      if (delta !== 0) return delta;
+    }
+    return 0;
+  })[candidates.length - 1];
+  return winner?.value;
+};
+
+/** Replaces every `var(--name, fallback)` with the property's value on this element. */
+const resolveVariables = (styles: PageStyles, element: Element, value: string, depth = 0): string | undefined => {
+  if (depth > 20) return undefined;
+  const start = value.indexOf('var(');
+  if (start === -1) return value;
+  let depthCount = 0;
+  let end = start + 3;
+  for (; end < value.length; end++) {
+    if (value[end] === '(') depthCount++;
+    if (value[end] === ')' && --depthCount === 0) break;
+  }
+  const [name, ...fallbackParts] = splitTopLevel(value.slice(start + 4, end));
+  const fallback = fallbackParts.length > 0 ? value.slice(start + 4, end).slice(value.slice(start + 4, end).indexOf(',') + 1).trim() : undefined;
+  const replacement = computedValue(styles, element, name.trim(), depth + 1) ?? (fallback !== undefined ? resolveVariables(styles, element, fallback, depth + 1) : undefined);
+  if (replacement === undefined) return undefined;
+  return resolveVariables(styles, element, value.slice(0, start) + replacement + value.slice(end + 1), depth + 1);
+};
+
+/** The value `property` ends up with on this element: cascade, inheritance and `var()` resolved. */
+export const computedValue = (
+  styles: PageStyles,
+  element: Element,
+  property: string,
+  depth = 0,
+): string | undefined => {
+  const inherits = INHERITED.has(property) || property.startsWith('--');
+  const cascaded = cascadedValue(styles, element, property);
+  if (cascaded === undefined || cascaded === 'inherit' || (cascaded === 'unset' && inherits)) {
+    return inherits && element.parentElement ? computedValue(styles, element.parentElement, property, depth) : undefined;
+  }
+  if (cascaded === 'initial' || cascaded === 'unset') return undefined;
+  return resolveVariables(styles, element, cascaded, depth);
+};

+// ---- the header look -------------------------------------------------------------------------------
+
+const toPx = (value: string | undefined): number | undefined => {
+  if (!value) return undefined;
+  const match = /^(-?[\d.]+)(px|rem)?$/.exec(value.trim());
+  if (!match) return undefined;
+  return parseFloat(match[1]) * (match[2] === 'rem' ? 16 : 1);
+};
+
+const toWeight = (value: string | undefined): number | undefined => {
+  if (!value) return undefined;
+  if (value === 'normal') return 400;
+  if (value === 'bold') return 700;
+  const weight = Number(value);
+  return Number.isFinite(weight) ? weight : undefined;
+};
+
+const hasNoExtraSpacing = (value: string | undefined): boolean => {
+  if (!value || value === 'normal') return value === 'normal';
+  const match = /^(-?[\d.]+)(px|em|rem)?$/.exec(value.trim());
+  if (!match) return false;
+  const amount = Math.abs(parseFloat(match[1]));
+  return match[2] === 'px' ? amount <= 0.2 : amount <= 0.0125;
+};
+
+export interface HeaderLook {
+  fontSizePx?: number;
+  fontWeight?: number;
+  textTransform?: string;
+  letterSpacing?: string;
+}
+
+export const headerLook = (styles: PageStyles, label: Element): HeaderLook => ({
+  fontSizePx: toPx(computedValue(styles, label, 'font-size')),
+  fontWeight: toWeight(computedValue(styles, label, 'font-weight')),
+  textTransform: computedValue(styles, label, 'text-transform'),
+  letterSpacing: computedValue(styles, label, 'letter-spacing'),
+});
+
+/** The design's header text: small (14px) text, semibold (medium to bold reads the same in the images). */
+export const isSmallSemibold = (look: HeaderLook) =>
+  look.fontSizePx !== undefined &&
+  look.fontSizePx >= 13 &&
+  look.fontSizePx <= 15 &&
+  look.fontWeight !== undefined &&
+  look.fontWeight >= 500 &&
+  look.fontWeight <= 700;
+
+/** The label keeps its own letter case and gets no extra letter spacing. */
+export const isSentenceCase = (look: HeaderLook) =>
+  (look.textTransform === undefined || look.textTransform === 'none') && hasNoExtraSpacing(look.letterSpacing ?? 'normal');
+
+const END = new Set(['flex-end', 'end', 'right']);
+
+/**
+ * Whether the label sits at the right edge of its header cell. Inside the grid's own label row the label is
+ * only as wide as its text, so only the row's justification or direction moves it; a header component that
+ * replaces the whole label row positions its label itself, by text alignment or a flex justification.
+ */
+export const isRightAligned = (styles: PageStyles, cell: Element, label: Element): boolean => {
+  const row = label.closest('.ag-header-cell-label');
+  if (row && cell.contains(row)) {
+    const justify = computedValue(styles, row, 'justify-content') ?? 'normal';
+    const reversed = (computedValue(styles, row, 'flex-direction') ?? 'row') === 'row-reverse';
+    return reversed ? !END.has(justify) : END.has(justify);
+  }
+  if (END.has(computedValue(styles, label, 'text-align') ?? '')) return true;
+  for (let node = label.parentElement; node && node !== cell.parentElement; node = node.parentElement) {
+    if (END.has(computedValue(styles, node, 'justify-content') ?? '')) return true;
+  }
+  return false;
+};
