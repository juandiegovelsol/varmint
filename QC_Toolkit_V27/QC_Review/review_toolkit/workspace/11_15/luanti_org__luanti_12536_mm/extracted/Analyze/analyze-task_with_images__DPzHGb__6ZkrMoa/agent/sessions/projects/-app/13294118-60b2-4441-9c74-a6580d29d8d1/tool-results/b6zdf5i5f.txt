+//
+// Every test builds a small piece of world (one map block plus its 26
+// neighbours, all air in full daylight), places a few nodes into it, runs the
+// engine's own MapblockMeshGenerator::generate() over the centre block exactly
+// like MapBlockMesh does, and then only looks at the geometry that comes out:

+// MeshMakeData reaches the game through a Client pointer, and the mesh
+// generator only ever calls ndef() on it: an inline, non-virtual IGameDef
+// helper that forwards to the virtual getNodeDefManager(). A Client cannot be
+// built headlessly at this point in the codebase's history, so this returns a
+// pointer that resolves -- through the very same Client -> IGameDef base
+// conversion the generator performs -- to the IGameDef above. No other part of
+// Client is ever touched.
+Client *clientFor(IGameDef *gamedef)
+{
+	alignas(Client) static unsigned char probe_storage[sizeof(Client)];
+	Client *probe = reinterpret_cast<Client *>(probe_storage);
+	ptrdiff_t offset = reinterpret_cast<char *>(static_cast<IGameDef *>(probe)) -
+			reinterpret_cast<char *>(probe);
+	return reinterpret_cast<Client *>(reinterpret_cast<char *>(gamedef) - offset);
+}

+aabb3f scaled(float x1, float y1, float z1, float x2, float y2, float z2)
+{
+	return aabb3f(x1 * BS, y1 * BS, z1 * BS, x2 * BS, y2 * BS, z2 * BS);

+// A wooden-fence-like connected node box: a post plus two rails per side.
+void makeFenceBox(NodeBox &box)
+{
+	box.type = NODEBOX_CONNECTED;
+	box.fixed = {scaled(-1/8.f, -1/2.f, -1/8.f, 1/8.f, 1/2.f, 1/8.f)};
+	NodeBoxConnected &c = box.getConnected();
+	for (float y : {3/16.f, -5/16.f}) {
+		c.connect_front.push_back(scaled(-1/16.f, y, -1/2.f, 1/16.f, y + 1/8.f, -1/8.f));
+		c.connect_left.push_back(scaled(-1/2.f, y, -1/16.f, -1/8.f, y + 1/8.f, 1/16.f));
+		c.connect_back.push_back(scaled(-1/16.f, y, 1/8.f, 1/16.f, y + 1/8.f, 1/2.f));
+		c.connect_right.push_back(scaled(1/8.f, y, -1/16.f, 1/2.f, y + 1/8.f, 1/16.f));
+	}
+}

+// The mesh used for the "mesh" drawtype node: an off-centre slab that only
+// covers the +X half of the node, so that every quarter turn is distinguishable.
+const aabb3f MESH_BOUNDS = scaled(0.f, -1/2.f, -1/16.f, 1/2.f, 1/2.f, 1/16.f);
+
+scene::SMesh *makeSlabMesh()
+{
+	const v3f a = MESH_BOUNDS.MinEdge, b = MESH_BOUNDS.MaxEdge;
+	const video::SColor white(255, 255, 255, 255);
+	video::S3DVertex v[8] = {
+		video::S3DVertex(a.X, a.Y, a.Z, 0, 0, -1, white, 0, 1),
+		video::S3DVertex(b.X, a.Y, a.Z, 0, 0, -1, white, 1, 1),
+		video::S3DVertex(b.X, b.Y, a.Z, 0, 0, -1, white, 1, 0),
+		video::S3DVertex(a.X, b.Y, a.Z, 0, 0, -1, white, 0, 0),
+		video::S3DVertex(a.X, a.Y, b.Z, 0, 0, 1, white, 1, 1),
+		video::S3DVertex(b.X, a.Y, b.Z, 0, 0, 1, white, 0, 1),
+		video::S3DVertex(b.X, b.Y, b.Z, 0, 0, 1, white, 0, 0),
+		video::S3DVertex(a.X, b.Y, b.Z, 0, 0, 1, white, 1, 0),
+	};
+	const u16 idx[36] = {
+		0, 2, 1, 0, 3, 2,  4, 5, 6, 4, 6, 7,  0, 1, 5, 0, 5, 4,
+		3, 6, 2, 3, 7, 6,  0, 4, 7, 0, 7, 3,  1, 2, 6, 1, 6, 5,
+	};
+	scene::SMeshBuffer *buf = new scene::SMeshBuffer();
+	buf->append(v, 8, idx, 36);
+	scene::SMesh *mesh = new scene::SMesh();
+	mesh->addMeshBuffer(buf);
+	buf->drop();
+	mesh->recalculateBoundingBox();
+	return mesh;
+}

+struct Extent {
+	v3f min_v, max_v;
+	size_t count = 0;
+
+	v3f center() const { return (min_v + max_v) * 0.5f; }
+	v3f half() const { return (max_v - min_v) * 0.5f; }
+};
+
+bool approx(float a, float b) { return std::fabs(a - b) < EPS; }
+
+bool sameBox(const Extent &e, const aabb3f &box)
+{
+	return e.count > 0 &&
+		approx(e.min_v.X, box.MinEdge.X) && approx(e.min_v.Y, box.MinEdge.Y) &&
+		approx(e.min_v.Z, box.MinEdge.Z) && approx(e.max_v.X, box.MaxEdge.X) &&
+		approx(e.max_v.Y, box.MaxEdge.Y) && approx(e.max_v.Z, box.MaxEdge.Z);
+}

+	// Runs the generator over block (0,0,0) and returns the extent of every
+	// vertex generated with the given texture, plus the vertices themselves.
+	Extent generate(u32 texture_id, std::vector<video::S3DVertex> *out = nullptr,
+			bool smooth_lighting = true)
+	{
+		MeshMakeData data(clientFor(&m_gamedef), false);
+		data.fillBlockDataBegin(v3s16(0, 0, 0));
+		std::vector<MapNode> air(MAP_BLOCKSIZE * MAP_BLOCKSIZE * MAP_BLOCKSIZE,
+				MapNode(CONTENT_AIR, LIGHT_SUN, 0));
+		for (s16 z = -1; z <= 1; z++)
+		for (s16 y = -1; y <= 1; y++)
+		for (s16 x = -1; x <= 1; x++)
+			data.fillBlockData(v3s16(x, y, z), air.data());
+		for (const auto &node : m_nodes)
+			data.m_vmanip.setNode(node.first, node.second);
+		data.setSmoothLighting(smooth_lighting);
+
+		MeshCollector collector;
+		NoopMeshManipulator meshmanip;
+		MapblockMeshGenerator(&data, &collector, &meshmanip).generate();
+
+		Extent e;
+		e.min_v = v3f(std::numeric_limits<float>::max());
+		e.max_v = v3f(-std::numeric_limits<float>::max());
+		for (const auto &buffers : collector.prebuffers)
+		for (const PreMeshBuffer &buf : buffers) {
+			if (buf.layer.texture_id != texture_id)
+				continue;
+			for (const video::S3DVertex &v : buf.vertices) {
+				e.min_v.X = std::min(e.min_v.X, v.Pos.X);
+				e.min_v.Y = std::min(e.min_v.Y, v.Pos.Y);
+				e.min_v.Z = std::min(e.min_v.Z, v.Pos.Z);
+				e.max_v.X = std::max(e.max_v.X, v.Pos.X);
+				e.max_v.Y = std::max(e.max_v.Y, v.Pos.Y);
+				e.max_v.Z = std::max(e.max_v.Z, v.Pos.Z);
+				e.count++;
+				if (out)
+					out->push_back(v);
+			}
+		}
+		return e;
+	}
+
+private:

+			const std::vector<content_t> &joins)
+	{
+		ContentFeatures f;
+		f.name = name;
+		f.drawtype = NDT_NODEBOX;
+		f.param_type = CPT_LIGHT;
+		f.solidness = 0;
+		makeFenceBox(f.node_box);
+		f.connects_to_ids = joins;
+		setAllTiles(f, texture_id);
+		return m_ndef->set(f.name, f);
+	}
+
+	NodeDefManager *m_ndef;
+	MeshGenGameDef m_gamedef;
+	scene::SMesh *m_slab_mesh;
+	std::vector<std::pair<v3s16, MapNode>> m_nodes;
+};

+std::string describe(const Extent &e)

+			e.count, e.min_v.X, e.min_v.Y, e.min_v.Z, e.max_v.X, e.max_v.Y, e.max_v.Z);
+	return buf;
+}

+// An "allfaces" node (the drawtype leaves use) with visual_scale = 2, deep
+// inside its map block: it must stay centred on its own node, twice as big.
+void TestMapblockMeshGenerator::testScaledAllfacesNodeOffset()
+{
+	const v3s16 p(13, 6, 11);
+	MeshGenScene scene;
+	scene.place(p, "test:leaves_scaled");
+	Extent e = scene.generate(TEX_LEAVES_SCALED);
+	const v3f expected = intToFloat(p, BS);
+
+	UTEST(e.count > 0 &&
+			approx(e.center().X, expected.X) && approx(e.center().Y, expected.Y) &&
+			approx(e.center().Z, expected.Z) &&
+			approx(e.half().X, BS) && approx(e.half().Y, BS) && approx(e.half().Z, BS),
+			"scaled allfaces node at (%d, %d, %d) is not centred on itself at "
+			"twice its size: %s", p.X, p.Y, p.Z, describe(e).c_str());
+}

# $JUNIT_OUT. The test ids (f2p + p2p from /tests/config.json) are in $TEST_IDS_FILE, one per line.
# Write plain shell commands here; no quoting of the whole block is needed.
run_tests() {

  # needs to pick up the new test_content_mapblock.cpp (added by test.patch)
  # and relink. CMake's Ninja generator reruns cmake's own configure step
  # automatically when it notices src/unittest/CMakeLists.txt changed.

  cat "$LOG" >&2

  # Luanti's legacy TestBase/TEST() framework only prints to stdout

  python3 - "$LOG" "$JUNIT_OUT" <<'PY'
import re
import sys
import xml.etree.ElementTree as ET

MODULE_RE = re.compile(r"^======== Testing module (\S+)")
TEST_RE = re.compile(r"^\[(PASS|FAIL)\] (\S+) - (\d+)ms$")

log_path, out_path = sys.argv[1], sys.argv[2]
with open(log_path, "r", errors="replace") as fh:
    lines = fh.read().splitlines()

cur_module = "Unknown"
pending = []
cases = []  # (classname, name, status, message)
for line in lines:
    m = MODULE_RE.match(line)
    if m:
        cur_module = m.group(1)
        pending = []
        continue
    m = TEST_RE.match(line)
    if m:
        status, name, _ms = m.groups()
        msg = "\n".join(pending).strip()
        cases.append((cur_module, name, status, msg))
        pending = []
        continue
    pending.append(line)

n_fail = sum(1 for c in cases if c[2] == "FAIL")
root = ET.Element("testsuite", name="luanti-unittests", tests=str(len(cases)),
                   failures=str(n_fail), errors="0")
for classname, name, status, msg in cases:
    tc = ET.SubElement(root, "testcase", classname=classname, name=name)
    if status == "FAIL":
        failure = ET.SubElement(tc, "failure",
                                 message=(msg.splitlines() or ["assertion failed"])[0][:200])
        failure.text = msg or "assertion failed"
ET.ElementTree(root).write(out_path, encoding="unicode", xml_declaration=True)
print(f"wrote {len(cases)} testcase(s), {n_fail} failing, to {out_path}", file=sys.stderr)
PY

#!/usr/bin/env python3
"""Turn the JUnit report into the task reward.

The reward is binary: 1.0 only when every configured f2p and p2p test is found exactly once in the
report and passed; otherwise 0.0. f2p_passed / f2p_total and p2p_passed / p2p_total record how close
a run came. The verifier reads test outcomes only, never the candidate's source.
"""
import html
import json
import os
import sys
import xml.etree.ElementTree as ET

CONF = json.load(open("/tests/config.json"))
F2P, P2P = CONF["f2p"], CONF["p2p"]
REPORT = os.environ.get("JUNIT_OUT", "/logs/verifier/junit.xml")
OUT_DIR = "/logs/verifier"

import atexit, json as _cj, os as _co, sys as _cs

_CANON_JSON = "/logs/verifier/reward.json"
_CANON_TXT = "/logs/verifier/reward.txt"
_canon = {}


def canon_reward(f2p_passed, f2p_total, p2p_passed, p2p_total, rubric):
    """THE reward policy for the whole corpus. BINARY -- there is no partial credit.

    tests_passed is 1.0 only when every f2p test passes AND every p2p test still passes. A
    strict subset of f2p passing is not partial progress worth scoring: the task is unsolved,
    so it is 0.0.

    reward is the same value, zeroed when the rubric judge failed a criterion. A "pass" or a
    "not graded" verdict never lowers the score -- the judge must never be able to penalise a
    run it could not grade.
    """
    f, t = int(f2p_passed), int(f2p_total)
    p, q = int(p2p_passed), int(p2p_total)
    solved = bool(t) and f == t and p == q
    tests_passed = 1.0 if solved else 0.0
    _canon.clear()
    _canon.update({
        "reward": 0.0 if rubric == "fail" else tests_passed,
        "tests_passed": tests_passed,
        "f2p_passed": f, "f2p_total": t,
        "p2p_passed": p, "p2p_total": q,
        "rubric_passed": {"pass": 1, "fail": 0}.get(rubric, -1),
    })
    return _canon["reward"]


@atexit.register
def _canon_flush():
    if not _canon:
        return
    try:
        _co.makedirs(_co.path.dirname(_CANON_JSON), exist_ok=True)
        with open(_CANON_JSON, "w") as _h:
            _cj.dump(_canon, _h, indent=2)
        # reward.txt too: harbor prefers reward.json, but the sandbox teardown script and the
        # lighthouse UI read reward.txt. Writing both from one dict means they cannot disagree.
        with open(_CANON_TXT, "w") as _h:
            _h.write(str(_canon["reward"]))
    except Exception as _exc:
        print("WARNING: could not write %s: %s" % (_CANON_JSON, _exc), file=_cs.stderr)


def rubric_verdict(argv):
    """The verdict test.sh hands in as `--rubric pass|fail|""`.

    Anything that is not an explicit pass/fail -- including the empty string test.sh passes
    when judge.py wrote no verdict -- means "not graded", which must never penalise the run.
    """
    if "--rubric" not in argv:
        return None
    i = argv.index("--rubric")
    value = argv[i + 1].strip().lower() if i + 1 < len(argv) else ""
    return value if value in ("pass", "fail") else None


def load_junit(path):
    """Every reported test, keyed by "classname name" and by name; a key seen twice with different
    outcomes counts as failed."""
    results = {}
    if not os.path.exists(path):
        return results
    try:
        root = ET.parse(path).getroot()
    except ET.ParseError:
        return results
    for tc in root.iter("testcase"):
        tags = {c.tag for c in tc}
        outcome = "skip" if "skipped" in tags else ("fail" if tags & {"failure", "error"} else "pass")
        full = f"{tc.get('classname', '')} {tc.get('name', '')}".strip()
        for key in {full, tc.get("name", "")}:
            if key in results and results[key] != outcome:
                results[key] = "fail"
            else:
                results[key] = outcome
    return results


def junit_key(test_id):
    """A pytest node id (tests/test_x.py::TestA::test_b[png]) as the JUnit report names it
    ("tests.test_x.TestA test_b[png]"); other ids unchanged."""
    if "::" not in test_id:
        return test_id
    parts = test_id.split("::")
    mod = parts[0][:-3] if parts[0].endswith(".py") else parts[0]
    return f"{'.'.join([mod.replace('/', '.')] + parts[1:-1])} {parts[-1]}"


def resolve(test_id, results):
    """The outcome of one configured test: an exact key match first (the id as written, or a
    pytest node id in JUnit form), else the single reported test whose key ends with it. Missing
    or ambiguous counts as not passed."""
    keys = [test_id, junit_key(test_id)]
    if " :: " in test_id:  # "file :: describe › test" (playwright list output)
        keys.append(test_id.replace(" :: ", " ", 1))
    if " > " in test_id:  # "describe > test": jest-junit joins with a space, playwright with " › "
        head, rest = test_id.split(" > ", 1)
        keys += [test_id.replace(" > ", " "), test_id.replace(" > ", " › "), head + " " + rest.replace(" > ", " › ")]
    keys += [k.replace('"', "") for k in keys]  # jest-junit drops quotation marks from names
    keys += [html.unescape(k) for k in keys]  # an id copied from a JUnit file may keep &apos; &quot;
    keys = list(dict.fromkeys(keys))
    for key in keys:
        if key in results:
            return results[key]
    for key in keys:
        hits = {k: v for k, v in results.items() if k.endswith(key)}
        if hits and len(set(hits.values())) == 1:
            return next(iter(hits.values()))
        if hits:
            return "ambiguous"
    # "file > test" with a describe level left out: a test in that file whose name ends the same
    parts = test_id.split(" > ")
    if len(parts) >= 2 and ("." in parts[0] or "/" in parts[0]):
        hits = {k: v for k, v in results.items() if k.startswith(parts[0] + " ") and k.endswith(parts[-1])}
        if hits and len(set(hits.values())) == 1:
            return next(iter(hits.values()))
        if hits:
            return "ambiguous"
    return "missing"


def main():
    results = load_junit(REPORT)
    detail = {}
    f2p_ok = 0
    for t in F2P:
        r = resolve(t, results)
        detail[f"f2p::{t}"] = r
        f2p_ok += r == "pass"
    p2p_ok = 0
    for t in P2P:
        r = resolve(t, results)
        detail[f"p2p::{t}"] = r
        p2p_ok += r == "pass"
    solved = bool(F2P) and f2p_ok == len(F2P) and p2p_ok == len(P2P)
    canon_reward(f2p_ok, len(F2P), p2p_ok, len(P2P), rubric_verdict(sys.argv[1:]))
    reward = dict(_canon)

    json.dump(reward, open(f"{OUT_DIR}/reward.json", "w"), indent=2)
    json.dump(detail, open(f"{OUT_DIR}/reward_detail.json", "w"), indent=2)
