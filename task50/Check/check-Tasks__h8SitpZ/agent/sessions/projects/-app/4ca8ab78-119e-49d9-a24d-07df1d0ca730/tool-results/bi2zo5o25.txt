=== instruction.md ===
# Screen does not communicate what was sent

When a withdrawal finishes, the screen should clearly show what happened on both sides — what was actually sent as well as what was received — regardless of how the amount was entered or whether the payout was unwrapped. If the information needed to work out what was sent isn't available yet, show the amount as entered rather than leaving it blank.

This is how it currently looks:

- `/app/.harbor_assets/bug_screenshot_1.png`
- `/app/.harbor_assets/bug_screenshot_2.png`

Fix it so the screen transparently reflects what actually happened, sent and received alike.

+    networkSymbol: NetworkSymbol;
+    flowType: YieldWithdrawFlowType;
+    completedAmount: string;
+    unwrappedAmount: string | null;
+    token: YieldFlowToken;
+    receiptToken: YieldFlowDisplayToken;
+    pricePerShareState: PricePerShareState;
+};

+type YieldWithdrawCompletedValues = {
+    input: YieldFlowCompleteValue;
+    output: YieldFlowCompleteValue;
+};

+/**
+ * Build the "sent → received" pair shown on the withdrawal-complete screen.
+ *
+ * A vault withdrawal always spends the vault receipt token (e.g. trSHETHp) and returns the
+ * underlying token (WETH) — or the native token (ETH) when the user unwraps. The completion screen
+ * must communicate that outgoing receipt token in every case (issue #30548), not only for the
+ * `redeem` unit.
+ *
+ * `completedAmount` is expressed in the input unit: shares for `redeem`, the underlying asset for
+ * `withdraw`. The receipt (shares) amount for the `withdraw` unit is therefore derived from the
+ * vault price-per-share, mirroring how the deposit flow derives its received receipt amount.
+ */
+export const getYieldWithdrawCompletedValues = ({
+    networkSymbol,
+    flowType,
+    completedAmount,
+    unwrappedAmount,
+    token,
+    receiptToken,
+    pricePerShareState,
+}: GetYieldWithdrawCompletedValuesParams): YieldWithdrawCompletedValues => {
+    const isSharesInput = flowType === 'redeem';
+
+    const sentReceiptAmount = isSharesInput
+        ? completedAmount
+        : (getWithdrawRequestAmount({
+              networkSymbol,
+              amount: completedAmount,
+              token,
+              receiptToken,
+              pricePerShare: pricePerShareState?.price,
+          }) ?? completedAmount);
+
+    const input: YieldFlowCompleteValue = {
+        token: receiptToken,
+        amount: sentReceiptAmount,
+    };
+
+    if (unwrappedAmount !== null) {
+        return {
+            input,
+            output: {
+                token: {
+                    networkSymbol,
+                    symbol: getNetworkDisplaySymbol(networkSymbol),
+                    decimals: getNetwork(networkSymbol).decimals,
+                },
+                amount: unwrappedAmount,
+            },
+        };
+    }
+
+    return {
+        input,
+        output: {
+            token,
+            amount:
+                isSharesInput && pricePerShareState
+                    ? getConvertedOutputTokenBalanceToInputTokenAmount({
+                          networkSymbol,
+                          token,
+                          outputToken: receiptToken,
+                          outputTokenBalance: completedAmount,
+                          pricePerShareState,
+                      })
+                    : completedAmount,
+        },
+    };
+};

=== tests/verifier.py ===
#!/usr/bin/env python3
"""Reward computation for trezor__trezor-suite-30548-mm.

Parses the JUnit XML report test.sh writes (jest + jest-junit) and computes the reward from
config.json's grounded f2p/p2p identifiers. NEVER inspects the candidate's source -- only the
test report.

Identifier format is "<classname> > <name>", where test.sh's JEST_JUNIT_* templates set
classname = the test file path as jest reports it (workspace-relative) and
name = the full ` > `-joined describe/it chain (confirmed empirically against this repo's own
report.xml).

Reward policy: BINARY. reward and tests_passed are 1.0 only when every f2p test passes AND
every p2p test still passes; reward is additionally zeroed if the rubric judge failed a
criterion (--rubric fail). rubric_passed is 1 pass / 0 fail / -1 not graded. There is no
partial credit -- a strict subset of f2p passing is 0.0, not 0.5.
"""
import atexit
import json
import os
import sys
import xml.etree.ElementTree as ET

CONF = json.load(open('/tests/config.json'))
F2P = CONF['f2p']
P2P = CONF['p2p']
REPORT = '/logs/verifier/report.xml'
REWARD_JSON = '/logs/verifier/reward.json'
REWARD_TXT = '/logs/verifier/reward.txt'

_canon = {}


def canon_reward(f2p_passed, f2p_total, p2p_passed, p2p_total, rubric):
    """THE reward policy for the whole corpus. BINARY -- there is no partial credit."""
    f, t = int(f2p_passed), int(f2p_total)
    p, q = int(p2p_passed), int(p2p_total)
    solved = bool(t) and f == t and p == q
    tests_passed = 1.0 if solved else 0.0
    _canon.clear()
    _canon.update({
        "reward": 0.0 if rubric == "fail" else tests_passed,
        "tests_passed": tests_passed,
        "f2p_passed": f, "f2p_total": t,
        "p2p_passed": p, "p2p_total": q,
        "rubric_passed": {"pass": 1, "fail": 0}.get(rubric, -1),
    })
    return _canon["reward"]


@atexit.register
def _canon_flush():
    if not _canon:
        return
    try:
        os.makedirs(os.path.dirname(REWARD_JSON), exist_ok=True)
        with open(REWARD_JSON, 'w') as fh:
            json.dump(_canon, fh, indent=2)
        with open(REWARD_TXT, 'w') as fh:
            fh.write(str(_canon["reward"]))
    except Exception as exc:
        print("WARNING: could not write %s: %s" % (REWARD_JSON, exc), file=sys.stderr)


def rubric_verdict(argv):
    """The verdict test.sh hands in as `--rubric pass|fail|""`."""
    if "--rubric" not in argv:
        return None
    i = argv.index("--rubric")
    value = argv[i + 1].strip().lower() if i + 1 < len(argv) else ""
    return value if value in ("pass", "fail") else None


def load_results():
    """Return {"<classname> > <name>": "pass"|"fail"|"skip"} parsed from the jest JUnit report."""
    results = {}
    if not os.path.exists(REPORT):
        return results
    try:
        tree = ET.parse(REPORT)
    except ET.ParseError:
        return results
    for tc in tree.iter('testcase'):
        classname = tc.get('classname', '')
        name = tc.get('name', '')
        key = f"{classname} > {name}" if classname else name
        failed = any(c.tag in ('failure', 'error') for c in tc)
        skipped = any(c.tag == 'skipped' for c in tc)
        results[key] = 'skip' if skipped else ('fail' if failed else 'pass')
    return results


def match(idents, results):
    """Match each configured identifier to a reported test's status. Exact match first (our
    identifiers are literally `${classname} > ${name}` from the same report format), falling
    back to substring tolerance in case of minor formatting drift."""
    out = {}
    for ident in idents:
        if ident in results:
            out[ident] = results[ident]
            continue
        hit = next((v for k, v in results.items() if ident in k or k in ident), None)
        out[ident] = hit or 'missing'
    return out


def main():
    results = load_results()
    f2p = match(F2P, results)
    p2p = match(P2P, results)

    f2p_pass = [k for k, v in f2p.items() if v == 'pass']
    p2p_pass = [k for k, v in p2p.items() if v == 'pass']

    canon_reward(len(f2p_pass), len(F2P), len(p2p_pass), len(P2P),
                 rubric_verdict(sys.argv[1:]))
    # Per-test detail, flat scalar keys alongside the 7 canonical ones -- lets the gate confirm
    # each f2p test fails individually on an empty submission, not just the aggregate count.
    _canon.update({f"f2p::{k}": 1.0 if v == 'pass' else 0.0 for k, v in f2p.items()})
    _canon.update({f"p2p::{k}": 1.0 if v == 'pass' else 0.0 for k, v in p2p.items()})

    print(json.dumps({**_canon, 'f2p': f2p, 'p2p': p2p}, indent=2))


if __name__ == '__main__':
    main()

# ---------------------------------------------------------------------------------------------
# trezor/trezor-suite — reusable build recipe (DONOR).
#
# Retargeting to another base commit is a pure string substitution of the 40-hex SHA below
# (tools/harvest_m2.py::retarget_dockerfile does exactly that); NOTHING else in this file is
# task-specific. It installs the WHOLE yarn workspace set, so it works no matter which of the
# ~100 workspaces (packages/*, suite/*, suite-common/*, suite-native/*, networks/*/*) the fix
# under test lives in.
# ---------------------------------------------------------------------------------------------

# Node 24: the repo pins `"engines": {"node": "24"}` in the root package.json and `.nvmrc` says
# 24.17.0 at this base commit. Do NOT downgrade — several transitive deps (and yarn 4.14's own
# `npmMinimalAgeGate` machinery) require >= 20, and a too-old Node makes `yarn install` hard-fail
# with no reward at all. bookworm-slim keeps the layer small; python3 is needed by the verifier
# (verifier.py / judge.py), git by the harness reset step.
FROM node:24-bookworm-slim

RUN apt-get update && apt-get install -y --no-install-recommends \
    git \
    ca-certificates \
    python3 \
 && rm -rf /var/lib/apt/lists/*

# The repo pins yarn via `yarnPath: .yarn/releases/yarn-4.14.1.cjs` + `packageManager: yarn@4.14.1`.
# corepack is what turns the `yarn` shim into that pinned release. Node 24 still ships corepack;
# the fallback keeps the recipe alive on images where it was dropped.
RUN (corepack enable || (npm install -g corepack@latest && corepack enable)) \
 && corepack --version

WORKDIR /app

# Shallow-fetch just the base commit (a full clone of this monorepo is >1 GB of history), pin to
# it, then STRIP .git so the agent cannot read the fix PR / future history (anti-cheat). Re-init a
# single base commit + an immutable `harbor_base` tag so the verifier can restore the test files
# it owns; `git apply` still works without upstream history.
RUN git init -q /app \
 && cd /app \
 && git remote add origin https://github.com/trezor/trezor-suite.git \
 && git fetch --depth 1 origin 7156f2b3ebb3f7752ee21a831e94b3451a1c5584 \
 && git checkout 7156f2b3ebb3f7752ee21a831e94b3451a1c5584 -q \
 && rm -rf /app/.git \
 && git -C /app init -q \
 && git -C /app -c user.email=h@l -c user.name=h add -A \
 && git -C /app -c user.email=h@l -c user.name=h commit -q -m base \
 && git -C /app tag harbor_base

# Bake ALL deps at BUILD time — the build has internet, the agent's solve RUN does NOT
# (task.toml [agent] network_mode = "no-network"). The verifier RUN is granted public network
# (task.toml [verifier] network_mode = "public"), but deps are pre-baked here anyway so grading
# doesn't depend on the npm registry being reachable, and stays fast.
#
#   * `--immutable`      : install exactly what yarn.lock says; fails loudly on drift.
#   * `--mode=skip-build`: the root package.json marks ~19 packages `dependenciesMeta.built=true`
#                          (playwright browser bundles, detox, react-native-skia, better-sqlite3,
#                          usb, sharp, …). Those build steps download hundreds of MB of browser
#                          binaries and compile native toolkits that NO jest unit suite in this
#                          repo needs; skipping them cuts the build from ~30 min to a few minutes
#                          and avoids a class of flaky network fetches. @swc/core and esbuild —
#                          the two that jest actually touches — ship prebuilt per-platform
#                          optional dependencies (@swc/core-linux-*), so they still work.
#   * every workspace links via `workspace:*` and every @suite-common / @trezor package resolves
#     its `main` to `src/index`, so jest runs straight off TypeScript sources: there is NO
#     per-workspace build step to run here.
# A scratch cache folder is removed inside the same layer so the download cache never ships.
ENV YARN_ENABLE_GLOBAL_CACHE=false \
    YARN_CACHE_FOLDER=/tmp/yarn-cache \
    YARN_NPM_MINIMAL_AGE_GATE=0
RUN cd /app \
 && yarn install --immutable --mode=skip-build \
 && rm -rf /tmp/yarn-cache /root/.cache/node/corepack

# jest-junit (the JUnit reporter jest itself doesn't ship) is intentionally NOT installed here.
# It is test-only tooling needed solely for grading, so it is installed by tests/test.sh at
# verify time instead — nothing test-only is baked into the agent-visible image.

# Visual evidence, readable by the agent at /app/.harbor_assets/. tests/ and solution/ are
# intentionally NOT copied into the image — the harness supplies them at verify time.
RUN mkdir -p /logs /app/.harbor_assets
COPY assets/ /app/.harbor_assets/

CMD ["bash"]
